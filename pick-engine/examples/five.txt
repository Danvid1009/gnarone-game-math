// Pick One: 5 options, RTP 0.95. odds = [4.75,4.75,4.75,4.75,4.75]. u: one uniform in [0,1). Returns the winning option index.
function pickOutcome(u) {
  const cum = [0,0.2,0.4,0.6000000000000001,0.8,1];
  if (u >= 1) u = 1 - 1e-15;
  let i = 0; while (i < cum.length - 2 && cum[i + 1] <= u) i++;
  return i;
}
